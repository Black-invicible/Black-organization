document.addEventListener('DOMContentLoaded', () => { const forms = document.querySelectorAll('form[data-form-type]'); forms.forEach((form) => { form.addEventListener('submit', async (event) => { event.preventDefault(); const type = form.dataset.formType; const status = document.getElementById(`${type}Status`); const submitButton = form.querySelector('button[type="submit"]'); const payload = Object.fromEntries(new FormData(form).entries()); if (submitButton) { submitButton.disabled = true; submitButton.textContent = 'Saving...'; } if (status) { status.textContent = 'Saving...'; status.className = 'form-status'; } try { const endpoint = type === 'contact' ? '/api/contact' : type === 'volunteer' ? '/api/volunteer' : type === 'donation' ? '/api/donations' : '/api/registrations'; const response = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(payload) }); const data = await response.json(); if (!response.ok) { throw new Error(data.message || 'Request failed.'); } if (status) { status.textContent = data.message || 'Saved successfully.'; status.className = 'form-status success'; } form.reset(); } catch (error) { if (status) { status.textContent = error.message || 'Something went wrong.'; status.className = 'form-status error'; } } finally { if (submitButton) { submitButton.disabled = false; submitButton.textContent = form.dataset.formType === 'donation' ? 'Support the Cause' : form.dataset.formType === 'volunteer' ? 'Join as Volunteer' : form.dataset.formType === 'contact' ? 'Send Message' : 'Register Now'; } } }); }); const dashboard = document.getElementById('dashboardStats'); if (dashboard) { fetch('/api/stats') .then((response) => response.json()) .then((stats) => { dashboard.innerHTML = `
Registrations ${stats.registrations || 0}
Messages ${stats.contacts || 0}
Volunteers ${stats.volunteers || 0}
Donations ${stats.donations || 0}
`; }) .catch(() => { dashboard.innerHTML = '
StatsUnavailable
'; }); } });
